  <script>

    // 預設大學生常見午餐選項

    const defaultLunchList = [

      { name: "燒臘便當", budget: "normal" },

      { name: "學餐自助餐", budget: "cheap" },

      { name: "乾麵配滷蛋", budget: "cheap" },

      { name: "便利商店御飯糰", budget: "cheap" },

      { name: "健康水煮餐盒", budget: "normal" },

      { name: "麥當勞 1+1 / 漢堡", budget: "normal" },

      { name: "日式拉麵", budget: "luxury" },

      { name: "韓式豆腐鍋", budget: "normal" },

      { name: "牛肉麵", budget: "normal" },

      { name: "火鍋 / 麻辣燙", budget: "luxury" },

      { name: "水餃 10 顆 + 酸辣湯", budget: "cheap" },

      { name: "早午餐拼盤", budget: "luxury" },

      { name: "地獄選擇：再睡十分鐘不吃了", budget: "cheap" }

    ];


    // 初始化 LocalStorage 資料

    let lunchList = JSON.parse(localStorage.getItem('lunch_options')) || defaultLunchList;

    let currentBudget = 'all';

    let isRolling = false;

    let rerollCount = 0;


    const rejectJokes = [

      "說好不挑食的呢？再給你一次機會！",

      "這家明明很好吃！好啦幫你重抽～",

      "再抽就要上課遲到囉！",

      "最後一次！不准再賴皮了！"

    ];


    function saveToStorage() {

      localStorage.setItem('lunch_options', JSON.stringify(lunchList));

      renderFoodList();

    }


    function setFilter(budget) {

      currentBudget = budget;

      document.querySelectorAll('.filter-btn').forEach(btn => {

        if(btn.dataset.budget === budget) {

          btn.className = "filter-btn active text-xs px-2.5 py-1 rounded-full font-medium transition bg-orange-500 text-white";

        } else {

          btn.className = "filter-btn text-xs px-2.5 py-1 rounded-full font-medium transition bg-white text-gray-600 hover:bg-orange-100";

        }

      });

    }


    function getFilteredList() {

      if (currentBudget === 'all') return lunchList;

      return lunchList.filter(item => item.budget === currentBudget);

    }


    function drawLunch(isReject = false) {

      if (isRolling) return;

      const pool = getFilteredList();


      if (pool.length === 0) {

        alert("此篩選條件下沒有任何食物選項，請切換條件或新增食物！");

        return;

      }


      isRolling = true;

      const display = document.getElementById('slotDisplay');

      const hint = document.getElementById('slotHint');

      const btnText = document.getElementById('btnText');

      const rejectBtn = document.getElementById('rejectBtn');


      display.classList.add('slot-rolling');

      btnText.innerText = "正在決定命運...";

      rejectBtn.classList.add('hidden');


      if (isReject) {

        rerollCount++;

        hint.innerText = rejectJokes[rerollCount % rejectJokes.length];

      } else {

        hint.innerText = "命運輪盤轉動中...";

      }


      let speedCounter = 0;

      const interval = setInterval(() => {

        const tempItem = pool[Math.floor(Math.random() * pool.length)];

        display.innerText = tempItem.name;

        speedCounter++;

        if (speedCounter > 15) {

          clearInterval(interval);

          finishDraw(pool);

        }

      }, 90);

    }


    function finishDraw(pool) {

      const finalItem = pool[Math.floor(Math.random() * pool.length)];

      const display = document.getElementById('slotDisplay');

      const hint = document.getElementById('slotHint');

      const btnText = document.getElementById('btnText');

      const rejectBtn = document.getElementById('rejectBtn');


      display.classList.remove('slot-rolling');

      display.innerText = finalItem.name;

      hint.innerText = "🎉 命運為你選好了，就吃這個！";

      btnText.innerText = "重新抽籤";

      rejectBtn.classList.remove('hidden');

      isRolling = false;


      // 施放慶祝彩帶

      confetti({

        particleCount: 50,

        spread: 60,

        origin: { y: 0.65 }

      });

    }


    function toggleManagePanel() {

      const panel = document.getElementById('managePanel');

      const arrow = document.getElementById('toggleArrow');

      panel.classList.toggle('hidden');

      arrow.innerText = panel.classList.contains('hidden') ? '▼' : '▲';

    }


    function renderFoodList() {

      const listContainer = document.getElementById('foodList');

      const countSpan = document.getElementById('itemCount');

      listContainer.innerHTML = '';

      countSpan.innerText = lunchList.length;


      lunchList.forEach((item, index) => {

        const tag = document.createElement('div');

        tag.className = "flex items-center gap-1.5 bg-gray-100 hover:bg-gray-200 text-gray-700 text-xs px-3 py-1.5 rounded-full transition";

        tag.innerHTML = `

          <span>${item.name}</span>

          <button onclick="removeFood(${index})" class="text-gray-400 hover:text-red-500 font-bold ml-1">×</button>

        `;

        listContainer.appendChild(tag);

      });

    }


    function addCustomFood() {

      const nameInput = document.getElementById('newFoodName');

      const budgetSelect = document.getElementById('newFoodBudget');

      const name = nameInput.value.trim();


      if (!name) return;

      lunchList.push({ name, budget: budgetSelect.value });

      saveToStorage();

      nameInput.value = '';

    }


    function removeFood(index) {

      lunchList.splice(index, 1);

      saveToStorage();

    }


    // 初始化渲染

    renderFoodList();

  </script>
